Configuración del Backend & Código Fuente

Conecta tu Google Apps Script o copia los archivos requeridos para Vercel.

Al guardar una URL válida, la aplicación pasará automáticamente de utilizar los datos simulados a sincronizar datos reales con Google Sheets.

Copia este código y pégalo en tu proyecto de Google Apps Script (Editor de código). 
/**
 * BACKEND GOOGLE APPS SCRIPT - SPA & WELLNESS
 */
function doGet(e) {
  var action = e.parameter.action || 'readAll';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === 'readAll') {
    var servicios = sheetToObjects(ss.getSheetByName('Servicios'));
    var especialistas = sheetToObjects(ss.getSheetByName('Especialistas'));
    var citas = sheetToObjects(ss.getSheetByName('Citas'));
    
    return createJsonResponse({
      status: 'success',
      servicios: servicios,
      especialistas: especialistas,
      citas: citas
    });
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (data.action === 'createAppointment') {
      var sheetCitas = ss.getSheetByName('Citas');
      
      // Bloqueo / Validación de Solapamiento
      var citas = sheetToObjects(sheetCitas);
      var conflicto = citas.some(function(c) {
        return c.fecha === data.fecha && 
               c.hora_inicio === data.hora_inicio && 
               c.id_especialista === data.id_especialista && 
               c.estado_cita !== 'Cancelada';
      });
      
      if (conflicto) {
        return createJsonResponse({ status: 'error', message: 'El especialista ya tiene una cita reservada a esa hora.' });
      }
      
      var newId = 'CIT-' + Math.floor(1000 + Math.random() * 9000);
      sheetCitas.appendRow([
        newId,
        data.fecha,
        data.hora_inicio,
        data.hora_fin,
        data.cliente_nombre,
        data.cliente_telefono,
        data.cliente_email,
        data.id_servicio,
        data.id_especialista,
        data.monto,
        'Confirmada',
        new Date().toISOString()
      ]);
      
      return createJsonResponse({ status: 'success', id_cita: newId });
    }
  } catch (err) {
    return createJsonResponse({ status: 'error', message: err.toString() });
  }
}

function sheetToObjects(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
